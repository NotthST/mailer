import csv
import sys


def load_csv_by_citmid(path):
    """
    Load a CSV file and return a dict:
        { citmid_value: citmcrdb_value }
    """
    data = {}

    with open(path, newline='', encoding="utf-8-sig") as f:
        reader = csv.DictReader(f)

        # Make sure the needed columns exist
        required_columns = {"citmid", "citmcrdb"}
        missing = required_columns - set(reader.fieldnames or [])
        if missing:
            raise ValueError(f"{path} is missing columns: {', '.join(missing)}")

        for row in reader:
            citmid = (row["citmid"] or "").strip()
            citmcrdb = (row["citmcrdb"] or "").strip()

            if not citmid:
                # skip rows without an ID
                continue

            if citmid in data:
                # you can change this to overwrite if you want
                print(
                    f"Warning: duplicate citmid '{citmid}' in {path}, keeping first value.",
                    file=sys.stderr,
                )
                continue

            data[citmid] = citmcrdb

    return data


def compare_files(file1, file2):
    data1 = load_csv_by_citmid(file1)
    data2 = load_csv_by_citmid(file2)

    all_citmids = sorted(set(data1.keys()) | set(data2.keys()))

    for citmid in all_citmids:
        in_1 = citmid in data1
        in_2 = citmid in data2

        if in_1 and in_2:
            v1 = data1[citmid]
            v2 = data2[citmid]

            if v1 == v2:
                print(f"{citmid}: SAME (citmcrdb = {v1})")
            else:
                print(
                    f"{citmid}: DIFFERENT "
                    f"(citmcrdb in {file1} = {v1!r}, in {file2} = {v2!r})"
                )
        elif in_1 and not in_2:
            print(f"{citmid}: only exists in {file1}")
        elif in_2 and not in_1:
            print(f"{citmid}: only exists in {file2}")


if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python compare_csv.py file1.csv file2.csv")
        sys.exit(1)

    file1 = sys.argv[1]
    file2 = sys.argv[2]

    compare_files(file1, file2)
