import csv
import sys

# === CONFIGURATION ===
# Change these if your columns are different
CITMID_COL_LETTER = "R"      # column containing citmid
CITMCRDB_COL_LETTER = "I"    # column containing citmcrdb

HAS_HEADER = True            # set to False if your CSV has no header row
DELIMITER = ","              # if your CSV is ; separated, change to ";"


def col_letter_to_index(letter: str) -> int:
    """
    Convert Excel-style column letter (A, B, ..., Z, AA, AB, ...) to 0-based index.
    R -> 17, I -> 8, etc.
    """
    letter = letter.strip().upper()
    result = 0
    for ch in letter:
        if not ("A" <= ch <= "Z"):
            raise ValueError(f"Invalid column letter: {letter}")
        result = result * 26 + (ord(ch) - ord("A") + 1)
    return result - 1  # 0-based


CITMID_IDX = col_letter_to_index(CITMID_COL_LETTER)
CITMCRDB_IDX = col_letter_to_index(CITMCRDB_COL_LETTER)


def load_csv_by_columns(path: str) -> dict:
    """
    Load a CSV into a dict: {citmid_value: citmcrdb_value}
    using fixed column indices.
    """
    data = {}

    # Try Windows encoding first (Excel on Windows usually uses this)
    with open(path, newline="", encoding="cp1252") as f:
        reader = csv.reader(f, delimiter=DELIMITER)

        if HAS_HEADER:
            next(reader, None)  # skip header row

        for row_num, row in enumerate(reader, start=2 if HAS_HEADER else 1):
            # skip empty rows
            if not row:
                continue

            # make sure row has enough columns
            if len(row) <= max(CITMID_IDX, CITMCRDB_IDX):
                print(
                    f"Warning: row {row_num} in {path} has not enough columns, skipping.",
                    file=sys.stderr,
                )
                continue

            citmid = row[CITMID_IDX].strip()
            citmcrdb = row[CITMCRDB_IDX].strip()

            if not citmid:
                continue

            if citmid in data:
                print(
                    f"Warning: duplicate citmid '{citmid}' in {path}, keeping first value.",
                    file=sys.stderr,
                )
                continue

            data[citmid] = citmcrdb

    return data


def compare_files(file1: str, file2: str):
    data1 = load_csv_by_columns(file1)
    data2 = load_csv_by_columns(file2)

    all_keys = sorted(set(data1.keys()) | set(data2.keys()))

    for citmid in all_keys:
        in_1 = citmid in data1
        in_2 = citmid in data2

        if in_1 and in_2:
            v1 = data1[citmid]
            v2 = data2[citmid]
            if v1 == v2:
                print(f"{citmid}: SAME (citmcrdb = {v1})")
            else:
                print(
                    f"{citmid}: DIFFERENT "
                    f"(citmcrdb in {file1} = {v1!r}, in {file2} = {v2!r})"
                )
        elif in_1 and not in_2:
            print(f"{citmid}: only exists in {file1}")
        elif in_2 and not in_1:
            print(f"{citmid}: only exists in {file2}")


if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python compare_csv_by_columns.py file1.csv file2.csv")
        sys.exit(1)

    file1 = sys.argv[1]
    file2 = sys.argv[2]

    compare_files(file1, file2)
