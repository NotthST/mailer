import csv
import os
import sys
from decimal import Decimal, InvalidOperation
from collections import defaultdict

# =======================
# CONFIGURATION
# =======================

# --- file1 (new file) column names ---
FILE1_DELIMITER = ";"
FILE1_CASE_ID_COL = "case_id"
FILE1_STATUS_COL = "status"
FILE1_CURRENCY_COL = "currency"
FILE1_AMOUNT_COL = "amount"
FILE1_REF2_COL = "ref2"       # column G
FILE1_ACCOUNT_COL = "account" # column H

# Deduplication priority
STATUS_PRIORITY = {
    "Closed": 3,
    "Pre-matched": 2,
    "Open": 1,
}

# --- repo files (old structure) ---
REPO_DELIMITER = ";"
REPO_HAS_HEADER = True

REPO_ID_COL_LETTER = "R"      # citmid
REPO_AMOUNT_COL_LETTER = "H"  # citmam
REPO_ACCOUNT_COL_LETTER = "C" # cexanuacc
REPO_REF2_COL_LETTER = "M"    # citmrfthr1


# =======================
# UTILITIES
# =======================

def col_letter_to_index(letter: str) -> int:
    """Excel column letter -> 0-based index (A=0, B=1, ..., Z=25, AA=26, etc.)"""
    letter = letter.strip().upper()
    result = 0
    for ch in letter:
        if not ("A" <= ch <= "Z"):
            raise ValueError(f"Invalid column letter: {letter}")
        result = result * 26 + (ord(ch) - ord("A") + 1)
    return result - 1


# Convert repo column letters into indices
REPO_ID_IDX = col_letter_to_index(REPO_ID_COL_LETTER)
REPO_AMOUNT_IDX = col_letter_to_index(REPO_AMOUNT_COL_LETTER)
REPO_ACCOUNT_IDX = col_letter_to_index(REPO_ACCOUNT_COL_LETTER)
REPO_REF2_IDX = col_letter_to_index(REPO_REF2_COL_LETTER)
REPO_MAX_IDX = max(REPO_ID_IDX, REPO_AMOUNT_IDX, REPO_ACCOUNT_IDX, REPO_REF2_IDX)


def parse_amount(value: str):
    """
    Parse various number formats into Decimal.
    Example:
      '14511.44' → Decimal("14511.44")
      '14.511,44' → Decimal("14511.44")
      '100,000' → Decimal("100000")
    """
    if value is None:
        return None
    s = value.strip()
    if not s:
        return None

    # remove thousand separators/spaces
    s = s.replace(" ", "").replace("'", "")

    # If both separators exist, detect decimal separator
    if "," in s and "." in s:
        last_comma = s.rfind(",")
        last_dot = s.rfind(".")
        if last_comma > last_dot:
            s = s.replace(".", "")
            s = s.replace(",", ".")
        else:
            s = s.replace(",", "")
    elif "," in s:
        s = s.replace(",", ".")

    try:
        return Decimal(s)
    except InvalidOperation:
        return None


# =======================
# LOAD FILE1 (the NEW file)
# =======================

def load_file1(path: str):
    """
    Load file1 and deduplicate on case_id using:
    Closed > Pre-matched > Open
    """
    groups = defaultdict(list)

    with open(path, newline="", encoding="cp1252", errors="replace") as f:
        reader = csv.DictReader(f, delimiter=FILE1_DELIMITER)

        required = {
            FILE1_CASE_ID_COL,
            FILE1_STATUS_COL,
            FILE1_AMOUNT_COL,
            FILE1_REF2_COL,
            FILE1_ACCOUNT_COL,
        }
        missing = required - set(reader.fieldnames)
        if missing:
            raise ValueError(f"Missing required columns in file1: {missing}")

        for row in reader:
            cid = (row.get(FILE1_CASE_ID_COL) or "").strip()
            if cid:
                groups[cid].append(row)

    result = {}

    for cid, rows in groups.items():
        best_row = None
        best_score = -1

        for r in rows:
            status = (r.get(FILE1_STATUS_COL) or "").strip()
            score = STATUS_PRIORITY.get(status, 0)

            if score >= best_score:
                best_score = score
                best_row = r

        if not best_row:
            continue

        amount_raw = (best_row.get(FILE1_AMOUNT_COL) or "").strip()
        amount_num = parse_amount(amount_raw)

        result[cid] = {
            "case_id": cid,
            "status": (best_row.get(FILE1_STATUS_COL) or "").strip(),
            "currency": (best_row.get(FILE1_CURRENCY_COL, "")).strip(),
            "amount_raw": amount_raw,
            "amount_num": amount_num,
            "account": (best_row.get(FILE1_ACCOUNT_COL, "")).strip(),
            "ref2": (best_row.get(FILE1_REF2_COL, "")).strip(),
        }

    return result


# =======================
# LOAD A REPO CSV FILE
# =======================

def load_repo_file(path: str):
    """
    Load one repo file using fixed column letters.
    Returns a dict: { citmid : {...values...} }
    """
    data = {}

    with open(path, newline="", encoding="cp1252", errors="replace") as f:
        reader = csv.reader(f, delimiter=REPO_DELIMITER)

        if REPO_HAS_HEADER:
            next(reader, None)

        for row in reader:
            if len(row) <= REPO_MAX_IDX:
                continue

            citmid = row[REPO_ID_IDX].strip()
            if not citmid:
                continue

            amount_raw = row[REPO_AMOUNT_IDX].strip()
            amount_num = parse_amount(amount_raw)
            account = row[REPO_ACCOUNT_IDX].strip()
            ref2 = row[REPO_REF2_IDX].strip()

            if citmid in data:
                continue

            data[citmid] = {
                "amount_raw": amount_raw,
                "amount_num": amount_num,
                "account": account,
                "ref2": ref2,
            }

    return data


# =======================
# MAIN COMPARISON
# =======================

def compare_file1_with_repo(file1_path: str, repo_folder: str):
    file1_data = load_file1(file1_path)

    # Load ALL repo files
    repo_files = {}
    for name in os.listdir(repo_folder):
        if name.lower().endswith(".csv"):
            full = os.path.join(repo_folder, name)
            if os.path.abspath(full) != os.path.abspath(file1_path):
                repo_files[name] = load_repo_file(full)

    results = []

    for cid, f1 in file1_data.items():

        f1_amount_raw = f1["amount_raw"]
        f1_amount_num = f1["amount_num"]
        f1_account = f1["account"]
        f1_ref2 = f1["ref2"]

        found = []  # list of (filename, values)

        for fname, mapping in repo_files.items():
            if cid in mapping:
                found.append((fname, mapping[cid]))

        if not found:
            results.append([
                cid, f1["status"],
                f1_amount_raw, "NOT_FOUND", "", "NOT_FOUND",
                f1_account, "NOT_FOUND", "", "NOT_FOUND",
                f1_ref2, "NOT_FOUND", "", "NOT_FOUND",
                ""
            ])
            continue

        # Collect values across files
        amount_values = []
        amount_sources = []
        account_values = []
        ref2_values = []

        # Add file1
        amount_values.append(f1_amount_num)
        amount_sources.append(f"file1:{f1_amount_raw}")
        account_values.append(f1_account)
        ref2_values.append(f1_ref2)

        files_matched = []

        for fname, entry in found:
            files_matched.append(fname)

            amount_values.append(entry["amount_num"])
            amount_sources.append(f"{fname}:{entry['amount_raw']}")
            account_values.append(entry["account"])
            ref2_values.append(entry["ref2"])

        # Determine differences
        non_null_amounts = [v for v in amount_values if v is not None]
        amount_status = "SAME" if len(set(non_null_amounts)) <= 1 else "DIFFERENT"
        amount_distinct = len(set(non_null_amounts))

        non_empty_acc = [v for v in account_values if v]
        account_status = "SAME" if len(set(non_empty_acc)) <= 1 else "DIFFERENT"
        account_distinct = len(set(non_empty_acc))

        non_empty_ref2 = [v for v in ref2_values if v]
        ref2_status = "SAME" if len(set(non_empty_ref2)) <= 1 else "DIFFERENT"
        ref2_distinct = len(set(non_empty_ref2))

        results.append([
            cid, f1["status"],
            f1_amount_raw,
            " | ".join(amount_sources),
            amount_distinct,
            amount_status,
            f1_account,
            " | ".join(f"{fname}:{acc}" for fname, acc in zip(files_matched, account_values[1:])),
            account_distinct,
            account_status,
            f1_ref2,
            " | ".join(f"{fname}:{rf}" for fname, rf in zip(files_matched, ref2_values[1:])),
            ref2_distinct,
            ref2_status,
            ", ".join(files_matched),
        ])

    # Write output CSV
    out_file = "comparison_results.csv"
    with open(out_file, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f, delimiter=";")
        writer.writerow([
            "case_id", "status",
            "file1_amount", "all_amounts_with_sources", "amount_distinct_count", "amount_status",
            "file1_account", "all_accounts_with_sources", "account_distinct_count", "account_status",
            "file1_ref2", "all_ref2_with_sources", "ref2_distinct_count", "ref2_status",
            "matched_files"
        ])
        writer.writerows(results)

    print(f"✔ Comparison completed.")
    print(f"✔ Results saved in: {out_file}")


# =======================
# ENTRY POINT
# =======================

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python compare_cases.py file1.csv repo_folder")
        sys.exit(1)

    file1_path = sys.argv[1]
    repo_folder = sys.argv[2]

    compare_file1_with_repo(file1_path, repo_folder)
