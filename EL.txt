import csv
import sys

# ===== CONFIGURATION =====
# Excel column letters
ID_COL_LETTER = "R"       # citmid (ID)
AMOUNT_COL_LETTER = "H"   # citmam (amount)

HAS_HEADER = True         # Set to False if there is NO header line
DELIMITER = ";"           # IMPORTANT: Excel in FR/Europe usually uses ';'


def col_letter_to_index(letter: str) -> int:
    """
    Convert Excel-style column letter (A, B, ..., Z, AA, AB, ...) to 0-based index.
    Example: A -> 0, B -> 1, H -> 7, R -> 17
    """
    letter = letter.strip().upper()
    result = 0
    for ch in letter:
        if not ("A" <= ch <= "Z"):
            raise ValueError(f"Invalid column letter: {letter}")
        result = result * 26 + (ord(ch) - ord("A") + 1)
    return result - 1  # 0-based index


ID_IDX = col_letter_to_index(ID_COL_LETTER)
AMOUNT_IDX = col_letter_to_index(AMOUNT_COL_LETTER)
MAX_IDX = max(ID_IDX, AMOUNT_IDX)


def load_csv(path: str) -> dict:
    """
    Load a CSV into a dict: { citmid: citmam }
    using fixed column indices.
    """
    data = {}

    # Excel on Windows usually = cp1252
    with open(path, newline="", encoding="cp1252", errors="replace") as f:
        reader = csv.reader(f, delimiter=DELIMITER)

        if HAS_HEADER:
            next(reader, None)  # skip header line

        start_row = 2 if HAS_HEADER else 1
        for row_num, row in enumerate(reader, start=start_row):
            if not row:
                continue

            # SAFETY: never index a short row
            if len(row) <= MAX_IDX:
                # Uncomment this if you want less spam:
                # if row_num <= 20 or row_num % 1000 == 0:
                print(
                    f"Warning: row {row_num} in {path} has not enough columns "
                    f"(len={len(row)}), skipping.",
                    file=sys.stderr,
                )
                continue

            citmid = row[ID_IDX].strip()
            citmam = row[AMOUNT_IDX].strip()

            if not citmid:
                continue

            if citmid in data:
                print(
                    f"Warning: duplicate citmid '{citmid}' in {path}, keeping first value.",
                    file=sys.stderr,
                )
                continue

            data[citmid] = citmam

    return data


def compare_files(file1: str, file2: str):
    data1 = load_csv(file1)
    data2 = load_csv(file2)

    all_ids = sorted(set(data1.keys()) | set(data2.keys()))

    for citmid in all_ids:
        in_1 = citmid in data1
        in_2 = citmid in data2

        if in_1 and in_2:
            v1 = data1[citmid]
            v2 = data2[citmid]
            if v1 == v2:
                print(f"{citmid}: SAME (citmam = {v1})")
            else:
                print(
                    f"{citmid}: DIFFERENT "
                    f"(citmam in {file1} = {v1!r}, in {file2} = {v2!r})"
                )
        elif in_1 and not in_2:
            print(f"{citmid}: only exists in {file1}")
        elif in_2 and not in_1:
            print(f"{citmid}: only exists in {file2}")


if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python import_csv.py file1.csv file2.csv")
        sys.exit(1)

    file1 = sys.argv[1]
    file2 = sys.argv[2]

    compare_files(file1, file2)
