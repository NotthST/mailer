PATTERN_REGEX="$(printf '%s|' "${patterns[@]}" | sed 's/|$//')"; find . -type f -print0 | xargs -0 -n1 bash -c 'f="$0"; case "$f" in *.tar.gz|*.tgz) tar -xzOf "$f" 2>/dev/null | grep -HnE --label="$f" "'"$PATTERN_REGEX"'" ;; *.gz) zgrep -HnE "'"$PATTERN_REGEX"'" "$f" 2>/dev/null ;; *.zip) unzip -p "$f" 2>/dev/null | grep -HnE --label="$f" "'"$PATTERN_REGEX"'" ;; *) grep -HnE --binary-files=without-match "'"$PATTERN_REGEX"'" "$f" 2>/dev/null ;; esac'
